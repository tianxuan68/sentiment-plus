async def analyze_review(text: str, top_n: int = 5) -> AnalyzeResult:
    """单条：并行调 sentiment + keywords + aspect（Mock 或真实）。
    入参:
        text: 评论文本，strip 后 len≥1
        top_n: 关键词 Top-N，1~20，默认 5
    返回 AnalyzeResult:
        sentiment: { label: 0|1, label_text, prob }
        keywords: [{ word, score }, ...]
        aspects: [{ aspect, polarity, polarity_text, prob }, ...]
        entities: [{ aspect, text, start, end, polarity, polarity_text }, ...]
        active_model: str          # 默认 "BERT"
        source: "mock"|"ai"|"mock_fallback"
        latency_ms: float | None
        text_preview: str | None
    HTTP: POST /analyze  Body: AnalyzeRequest { text, top_n? }
    """

    """返回: { satisfaction_score, avg_rating, review_count, positive_ratio,
              trend_direction: up|down|stable, source }
    HTTP: GET /stats/summary
    """

async def get_rating_distribution() -> RatingDistribution:
    """返回: { buckets: [{rating, count, ratio}], avg_rating, total, source }
    HTTP: GET /stats/rating

    """返回: { pros: [{word, score}], cons: [...], source }

    """返回: { metrics: [{model, acc, f1, owner}], best_model,
