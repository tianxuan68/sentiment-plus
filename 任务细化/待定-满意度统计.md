# 三、函数级接口契约

> 文件：`analytics/stats.py`、`analytics/report.py`、`analytics/api.py`

```python
def load_reviews(
    path: str = "data/processed/unified_reviews.csv",
) -> pd.DataFrame:
    """上游: 杨国东 build_unified_reviews()
    必需列: text, label, overall, review_time, lang
    """

def compute_polarity(
    df: pd.DataFrame,
    label_col: str = "label",
) -> dict:
    """正负比例。
    入参: df — unified_reviews；label_col — 0/1 列名
    返回: {
        "positive": int,
        "negative": int,
        "total": int,
        "positive_ratio": float,    # 0~1
        "negative_ratio": float,
    }
    下游: 郑平高 get_polarity_stats() → PolarityStat
    """

def compute_rating_distribution(
    df: pd.DataFrame,
    rating_col: str = "overall",
    map_to_1_5: bool = True,
) -> dict:
    """评分分布（英文 overall 0~4 可映射 1~5 星）。
    返回: {
        "buckets": [{ "rating": int, "count": int, "ratio": float }, ...],
        "avg_rating": float,
        "total": int,
    }
    下游: 郑平高 get_rating_distribution() → RatingDistribution
    HTTP: GET /api/v1/analytics/rating_distribution
    """

def compute_trend(
    df: pd.DataFrame,
    time_col: str = "review_time",
    freq: str = "D",
) -> list[dict]:
    """按日/周聚合评论量。
    返回: [{ "name": str, "value": int, "positive": int, "negative": int }, ...]
    下游: 郑平高 get_trend_stats() → list[TrendPoint]
    """

def build_satisfaction_report(
    df: pd.DataFrame,
    output_path: str = "data/processed/satisfaction_report.json",
) -> dict:
    """汇总 KPI 写 JSON。
    返回: {
        "satisfaction_score": float,     # 0~100，如 positive_ratio*100
        "avg_rating": float,
        "review_count": int,
        "positive_ratio": float,
        "trend_direction": "up"|"down"|"stable",
        "polarity": compute_polarity(...),
        "rating": compute_rating_distribution(...),
    }
    下游: 郑平高 get_dashboard_summary()
    """
```

### HTTP（sentiment-ai · 交 1 号）

| 方法 | 路径 | 请求 | 响应 `result` |
| ---- | ---- | ---- | ------------- |
| GET | `/api/v1/analytics/rating_distribution` | — | `RatingDistribution`（见郑平高 schema） |
| GET | `/api/v1/analytics/polarity` | — | 同 `compute_polarity()` 返回 |
| GET | `/api/v1/analytics/trend` | `?freq=D` | `[TrendPoint, ...]` |
